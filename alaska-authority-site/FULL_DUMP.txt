===== build.js =====
/**
 * Authority Site Builder (v1 – clean canonical)
 *
 * Input:  data/ports/*.json
 * Output: dist/ (static HTML)
 */

import fs from "fs";
import path from "path";

/* ---------------- config ---------------- */

const ROOT = process.cwd();
const DIST = path.join(ROOT, "dist");

const SITE = {
  origin: "http://localhost:8787",
  name: "Alaska Cruise Port Authority",
};

/* ---------------- utils ---------------- */

function read(file) {
  return fs.readFileSync(file, "utf8");
}

function write(file, contents) {
  fs.mkdirSync(path.dirname(file), { recursive: true });
  fs.writeFileSync(file, contents, "utf8");
}

function cleanDist() {
  if (fs.existsSync(DIST)) fs.rmSync(DIST, { recursive: true, force: true });
  fs.mkdirSync(DIST, { recursive: true });
}

function copyDir(src, dest) {
  if (!fs.existsSync(src)) return;
  fs.mkdirSync(dest, { recursive: true });
  for (const entry of fs.readdirSync(src)) {
    const s = path.join(src, entry);
    const d = path.join(dest, entry);
    if (fs.statSync(s).isDirectory()) {
      copyDir(s, d);
    } else {
      fs.copyFileSync(s, d);
    }
  }
}

function esc(s = "") {
  return String(s)
    .replaceAll("&", "&amp;")
    .replaceAll("<", "&lt;")
    .replaceAll(">", "&gt;");
}

function render(template, vars) {
  let out = template;
  for (const [key, value] of Object.entries(vars)) {
    out = out.replaceAll(`{{${key}}}`, value ?? "");
  }
  return out;
}

/* ---------------- data ---------------- */

function loadPorts() {
  const dir = path.join(ROOT, "data", "ports");
  if (!fs.existsSync(dir)) return [];
  return fs.readdirSync(dir)
    .filter(f => f.endsWith(".json"))
    .map(f => JSON.parse(read(path.join(dir, f))));
}

/* ---------------- templates ---------------- */

const BASE = read(path.join(ROOT, "templates", "base.html"));

/* ---------------- build ---------------- */

function build() {
  cleanDist();

  // copy static assets (css, images, etc)
  copyDir(path.join(ROOT, "public"), DIST);

  const ports = loadPorts();
  const urls = [];

  /* ---------- homepage ---------- */

  const homeHtml = render(BASE, {
    PAGE_TITLE: SITE.name,
    META_DESC: "Independent feasibility guidance for Alaska cruise ports.",
    CANONICAL: SITE.origin + "/",
    NAV: `<a href="/">Home</a> <a href="/ports/">Ports</a>`,
    JSON_LD: "",
    CONTENT: `
      <h1>${SITE.name}</h1>
      <p>Independent feasibility guidance for Alaska cruise ports.</p>
      <ul>
        ${ports.map(p =>
          `<li><a href="/ports/${p.port_id}/">${esc(p.port_name)}</a></li>`
        ).join("")}
      </ul>
    `,
  });

  write(path.join(DIST, "index.html"), homeHtml);
  urls.push("/");

  /* ---------- ports index ---------- */

  const portsIndex = render(BASE, {
    PAGE_TITLE: `Ports | ${SITE.name}`,
    META_DESC: "Alaska cruise port guides.",
    CANONICAL: SITE.origin + "/ports/",
    NAV: `<a href="/">Home</a> <a href="/ports/">Ports</a>`,
    JSON_LD: "",
    CONTENT: `
      <h1>Ports</h1>
      <ul>
        ${ports.map(p =>
          `<li><a href="/ports/${p.port_id}/">${esc(p.port_name)}</a></li>`
        ).join("")}
      </ul>
    `,
  });

  write(path.join(DIST, "ports", "index.html"), portsIndex);
  urls.push("/ports/");

  /* ---------- individual ports ---------- */

  for (const port of ports) {
    const portHtml = render(BASE, {
      PAGE_TITLE: `${port.port_name} | ${SITE.name}`,
      META_DESC: port.meta_description || "",
      CANONICAL: SITE.origin + `/ports/${port.port_id}/`,
      NAV: `
        <a href="/">Home</a>
        <a href="/ports/">Ports</a>
        <a href="/ports/${port.port_id}/">${esc(port.port_name)}</a>
      `,
      JSON_LD: "",
      CONTENT: `
        <h1>${esc(port.port_name)}</h1>
        <p>${esc(port.meta_description || "")}</p>
        <ul>
          <li><a href="/ports/${port.port_id}/tours/">Tours</a></li>
          <li><a href="/ports/${port.port_id}/cruise-ships-2026/">Cruise Ships 2026</a></li>
        </ul>
      `,
    });

    write(
      path.join(DIST, "ports", port.port_id, "index.html"),
      portHtml
    );

    urls.push(`/ports/${port.port_id}/`);
  }

  console.log(`✅ Built ${urls.length} pages`);
}

/* ---------------- run ---------------- */

build();
